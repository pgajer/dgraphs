cmake_minimum_required(VERSION 3.20)
project(ian_external_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ian CONFIG REQUIRED)
add_executable(ian_consumer main.cpp)
target_link_libraries(ian_consumer PRIVATE ian::core)
if(R_INCLUDE AND R_LIBRARY AND R_BRIDGE_SOURCE)
  add_library(ian_bridge MODULE ${R_BRIDGE_SOURCE})
  target_include_directories(ian_bridge PRIVATE ${R_INCLUDE})
  target_link_libraries(ian_bridge PRIVATE ian::core ${R_LIBRARY})
  set_target_properties(ian_bridge PROPERTIES PREFIX "" SUFFIX ".so")
endif()
