cmake_minimum_required(VERSION 3.13)
project(ian_sequence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CLARABEL_SOURCE OR NOT CLARABEL_LIBRARY)
 message(FATAL_ERROR "Set pinned CLARABEL_SOURCE and CLARABEL_LIBRARY")
endif()
add_executable(ian_sequence sequence.cpp)
target_include_directories(ian_sequence PRIVATE ${CLARABEL_SOURCE}/include)
target_link_libraries(ian_sequence PRIVATE ${CLARABEL_LIBRARY})
target_compile_options(ian_sequence PRIVATE -Wall -Wextra -Wpedantic)
