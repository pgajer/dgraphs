cmake_minimum_required(VERSION 3.13)
project(ian_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CLARABEL_SOURCE OR NOT CLARABEL_LIBRARY OR NOT JSON_INCLUDE)
 message(FATAL_ERROR "Set CLARABEL_SOURCE, CLARABEL_LIBRARY and JSON_INCLUDE")
endif()
set(HASHES "")
foreach(FILE engine.cpp engine.hpp checkpoint.hpp input.hpp stages.hpp numeric.hpp solver.hpp CMakeLists.txt)
 file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/${FILE} HASH)
 string(APPEND HASHES ${HASH})
endforeach()
string(SHA256 SOURCE_HASH ${HASHES})
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/config.json CONFIG_HASH)
add_executable(ian_engine engine.cpp)
target_include_directories(ian_engine PRIVATE ${CLARABEL_SOURCE}/include ${JSON_INCLUDE})
target_link_libraries(ian_engine PRIVATE ${CLARABEL_LIBRARY})
target_compile_options(ian_engine PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_compile_definitions(ian_engine PRIVATE SOURCE_HASH="${SOURCE_HASH}" CONFIG_HASH="${CONFIG_HASH}")
