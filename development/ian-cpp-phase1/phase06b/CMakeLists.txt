cmake_minimum_required(VERSION 3.20)
project(ian_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CLARABEL_SOURCE OR NOT CLARABEL_LIBRARY OR NOT JSON_INCLUDE)
  message(FATAL_ERROR "Set CLARABEL_SOURCE, CLARABEL_LIBRARY and JSON_INCLUDE")
endif()
file(GLOB_RECURSE IDENTITY_FILES CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} include/*.hpp src/*.hpp src/*.cpp src/*.inc tests/*.cpp)
list(SORT IDENTITY_FILES)
set(HASHES "")
foreach(FILE IN LISTS IDENTITY_FILES)
  file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/${FILE} HASH)
  string(APPEND HASHES ${HASH})
endforeach()
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt CMAKE_HASH)
string(SHA256 SOURCE_HASH "${HASHES}${CMAKE_HASH}")
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/config.json CONFIG_HASH)
add_library(ian::backend SHARED IMPORTED)
set_target_properties(ian::backend PROPERTIES IMPORTED_LOCATION "${CLARABEL_LIBRARY}")
add_library(ian_core STATIC src/core.cpp)
add_library(ian::core ALIAS ian_core)
set_target_properties(ian_core PROPERTIES EXPORT_NAME core)
target_include_directories(ian_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
  PRIVATE ${CLARABEL_SOURCE}/include ${JSON_INCLUDE} src)
target_link_libraries(ian_core PUBLIC ian::backend)
target_compile_definitions(ian_core PRIVATE SOURCE_HASH="${SOURCE_HASH}" CONFIG_HASH="${CONFIG_HASH}")
target_compile_options(ian_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(ian_engine src/cli.cpp)
add_executable(ian_probe tests/probe.cpp)
add_executable(ian_checkpoint_tool tests/checkpoint_tool.cpp)
foreach(TARGET ian_engine ian_probe ian_checkpoint_tool)
  target_include_directories(${TARGET} PRIVATE src ${JSON_INCLUDE})
  target_link_libraries(${TARGET} PRIVATE ian::core)
  target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endforeach()
if(R_INCLUDE AND R_LIBRARY)
  add_library(ian_bridge MODULE src/r_bridge.cpp)
  target_include_directories(ian_bridge PRIVATE ${R_INCLUDE})
  target_link_libraries(ian_bridge PRIVATE ian::core ${R_LIBRARY})
  set_target_properties(ian_bridge PROPERTIES PREFIX "" SUFFIX ".so")
  target_compile_options(ian_bridge PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
get_filename_component(BACKEND_NAME "${CLARABEL_LIBRARY}" NAME)
configure_package_config_file(cmake/ianConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ianConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ian)
install(TARGETS ian_core EXPORT ianTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION include)
install(FILES "${CLARABEL_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES IAN-LICENSE.txt NUMPY-LICENSE.txt config.json DESTINATION ${CMAKE_INSTALL_DATADIR}/ian)
install(EXPORT ianTargets NAMESPACE ian:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ian)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ianConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ian)
