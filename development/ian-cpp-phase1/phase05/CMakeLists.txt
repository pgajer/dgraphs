cmake_minimum_required(VERSION 3.13)
project(ian_sensitivity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CLARABEL_SOURCE OR NOT CLARABEL_LIBRARY OR NOT JSON_INCLUDE)
  message(FATAL_ERROR "Set CLARABEL_SOURCE, CLARABEL_LIBRARY and JSON_INCLUDE")
endif()
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp PROBE_HASH)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/../phase04/config.json CONFIG_HASH)
add_executable(ian_probe probe.cpp)
target_include_directories(ian_probe PRIVATE ${CLARABEL_SOURCE}/include ${JSON_INCLUDE})
target_link_libraries(ian_probe PRIVATE ${CLARABEL_LIBRARY})
target_compile_options(ian_probe PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_compile_definitions(ian_probe PRIVATE SOURCE_HASH="${PROBE_HASH}" CONFIG_HASH="${CONFIG_HASH}")
